package observer

import (
 "context"
 "io"
 "math"
 "net/http"
 "net/http/httptest"
 "path/filepath"
 "strings"
 "testing"

 deltav1 "github.com/Seyamalam/digital-delta/services/node/gen/digitaldelta/v1"
)

func TestReviewNonFiniteEventPoisonsReplay(t *testing.T) {
 store, err := OpenStore(filepath.Join(t.TempDir(), "observer.db")); if err != nil { t.Fatal(err) }; defer store.Close()
 hub := NewHub(store)
 service := NewService(hub)
 _, err = service.Publish(context.Background(), &deltav1.ObserverServicePublishRequest{
  SourceNodeId:"field-n4", Event:&deltav1.DomainEvent{EventId:"bad", Body:&deltav1.DomainEvent_EdgeRiskPredicted{EdgeRiskPredicted:&deltav1.EdgeRiskPredicted{EdgeId:"E3",Probability:float32(math.NaN())}}},
 }); if err != nil { t.Fatal(err) }
 _, err = service.Publish(context.Background(), &deltav1.ObserverServicePublishRequest{SourceNodeId:"field-n4", Event:&deltav1.DomainEvent{EventId:"good"}}); if err != nil { t.Fatal(err) }
 handler := NewHTTPHandler(hub,nil)
 for attempt:=0;attempt<2;attempt++ {
  recorder:=httptest.NewRecorder()
  handler.ServeHTTP(recorder, httptest.NewRequest(http.MethodGet,"/observer/events",nil))
  body,_:=io.ReadAll(recorder.Result().Body)
  if recorder.Code!=200 || strings.Contains(string(body),"good") || len(body)!=0 {t.Fatalf("unexpected response %d %s",recorder.Code,body)}
  t.Logf("attempt %d: HTTP 200, %d SSE bytes, valid sequence 2 permanently blocked",attempt,len(body))
 }
}
